import { useEffect, useRef, useState, useCallback } from 'react';

// Custom WebRTC Stream hook engineered for Daruk Video Conferencing
export interface PeerConnectionConfig {
  roomId: string;
  iceServers: RTCIceServer[];
  onTrackAdded: (stream: MediaStream, peerId: string) => void;
}

export const useWebRTCStream = ({ roomId, iceServers, onTrackAdded }: PeerConnectionConfig) => {
  const peerConnections = useRef<Map<string, RTCPeerConnection>>(new Map());
  const [isConnecting, setIsConnecting] = useState(true);

  const createPeerConnection = useCallback((peerId: string) => {
    const pc = new RTCPeerConnection({ iceServers });

    pc.ontrack = (event) => {
      onTrackAdded(event.streams[0], peerId);
    };

    peerConnections.current.set(peerId, pc);
    return pc;
  }, [iceServers, onTrackAdded]);

  return { createPeerConnection, peerConnections: peerConnections.current, isConnecting };
};
